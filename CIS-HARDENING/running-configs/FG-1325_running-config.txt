# FG-1325 - FortiGate 7.0.9 (FortiGate-VM64-KVM)
# Secciones configuradas en la practica, obtenidas con 'show' en la consola.
# La configuracion base se hizo desde la GUI (webterm, https://172.13.25.1),
# salvo la IP de port2, que se puso por consola para poder abrir la GUI.

FG-1325 # show system global
config system global
    set alias "FortiGate-VM64-KVM"
    set hostname "FG-1325"
    set pre-login-banner enable
    set timezone 04
end

FG-1325 # show system interface port1
config system interface
    edit "port1"
        set vdom "root"
        set mode dhcp
        set allowaccess ping https ssh
        set type physical
        set snmp-index 1
    next
end

FG-1325 # show system interface port2
config system interface
    edit "port2"
        set vdom "root"
        set ip 172.13.25.1 255.255.255.252
        set allowaccess ping https ssh
        set type physical
        set alias "HACIA-R1-1325"
        set device-identification enable
        set lldp-transmission enable
        set role lan
        set snmp-index 2
    next
end

FG-1325 # show router static
config router static
    edit 1
        set dst 10.13.25.0 255.255.255.0
        set gateway 172.13.25.2
        set device "port2"
    next
end

FG-1325 # show firewall policy
config firewall policy
    edit 1
        set name "LAN-a-INTERNET"
        set uuid 70a86bde-dede-51f1-109d-252451adb6c5
        set srcintf "port2"
        set dstintf "port1"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
        set nat enable
        set tcp-mss-sender 1360
        set tcp-mss-receiver 1360
    next
end
